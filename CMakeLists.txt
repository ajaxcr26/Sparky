cmake_minimum_required(VERSION 3.22)
project(SPARK VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
option(SPARK_PLUGIN "Build JUCE VST3 and standalone" ON)
add_library(spark_engine src/Engine.cpp)
target_include_directories(spark_engine PUBLIC src)
set_target_properties(spark_engine PROPERTIES POSITION_INDEPENDENT_CODE ON)
enable_testing()
add_executable(spark_tests tests/EngineTests.cpp)
target_link_libraries(spark_tests PRIVATE spark_engine)
add_test(NAME engine COMMAND spark_tests)
if(SPARK_PLUGIN)
 include(FetchContent)
 FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 51a8a6d7aeae7326956d747737ccf1575e61e209)
 FetchContent_MakeAvailable(JUCE)
 juce_add_plugin(SPARK COMPANY_NAME "SPARK" PLUGIN_MANUFACTURER_CODE Sprk PLUGIN_CODE Sp01 FORMATS VST3 Standalone PRODUCT_NAME "SPARK" IS_SYNTH TRUE NEEDS_MIDI_INPUT TRUE NEEDS_MIDI_OUTPUT TRUE COPY_PLUGIN_AFTER_BUILD FALSE)
 target_sources(SPARK PRIVATE src/Plugin.cpp)
 target_link_libraries(SPARK PRIVATE spark_engine juce::juce_audio_utils juce::juce_dsp)
 juce_add_console_app(spark_integration PRODUCT_NAME "SPARK Integration")
 target_sources(spark_integration PRIVATE tests/PluginTests.cpp)
 target_link_libraries(spark_integration PRIVATE spark_engine juce::juce_audio_utils juce::juce_dsp)
 target_compile_definitions(spark_integration PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
 add_test(NAME plugin_integration COMMAND spark_integration)
 target_compile_definitions(SPARK PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
endif()
